macOS / Linux

Windows (PowerShell)

iwr https://csoai.org/install.ps1 | iex -ArgumentList "--model","qwen3:30b-a3b","--name","my-sovereign-ai"

docker run -d --name sovereign-ai -p 8000:8000 -p 3101:3101 csoai/sov3:latest --model qwen3:30b-a3b
