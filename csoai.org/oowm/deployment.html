Deployment Surfaces

# Docker
docker pull csoai/oowm:latest
docker run -d -p 3101:3101 \
  -v ~/.sov3:/var/sov3 \
  --name sov3 csoai/oowm:latest

# Python pip
pip install sovereign-oowm
python -m sovereign_oowm.serve --port 3101

# Node.js / npm
npm install -g @csoai/oowm
npx sov3 serve --port 3101

# Apple Foundation Models Provider (iOS 26+, macOS 26+)
sov3 deploy --provider=apple-foundation-models
# Routes via Apple Intelligence on-device when available
# Falls back to sovereign runtime when not

# Sovereign TUI (keyboard-driven)
brew install csoai/tap/sov3-tui
sov3
