```python
import nacl.signing
import hashlib
import json

class SigilEmitter:
    def __init__(self, sovereign_key: bytes):
        self.key = nacl.signing.SigningKey(sovereign_key)
    
    def emit(self, op: str, actor: str, target: str, message: str) -> dict:
        """Emit a sovereign SIGIL with Ed25519 signature + SHA-256 hash."""
        timestamp = datetime.utcnow().isoformat()
        line = f"{op}|{actor}|{target}|{message}|{timestamp}"
        digest = hashlib.sha256(line.encode()).hexdigest()
        signature = self.key.sign(line.encode()).hex()
        return {
            'line': line,
            'digest': digest,
            'signature': signature,
            'verify_url': f'https://proofof.ai/verify/{digest}'
        }
    
    def emit_bft_proposal(self, charter: str, current_text: str, proposed_text: str) -> dict:
        message = f"BFT_PROPOSAL|{charter}|{current_text[:100]}|{proposed_text[:100]}"
```
