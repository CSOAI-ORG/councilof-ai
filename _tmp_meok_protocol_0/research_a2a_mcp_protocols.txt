2. Google A2A (Agent-to-Agent Protocol)
2.1 Overview
The Agent2Agent (A2A) Protocol is an open standard for communication and collaboration between independent AI agents. Originally developed by Google and donated to the Linux Foundation in June 2025, it is the leading protocol for cross-vendor, cross-domain agent interoperability.
2.2 Technical Architecture
Core Design Principles
Simple: Reuses HTTP, JSON-RPC 2.0, and Server-Sent Events
Enterprise Ready: Built-in authentication, authorization, privacy, tracing
Async First: Designed for long-running tasks and human-in-the-loop
Modality Agnostic: Supports text, audio/video, structured data, file references
Opaque Execution: Agents share neither internal state nor tool implementations
Transport & Communication
Primary Transport: JSON-RPC 2.0 over HTTP/HTTPS
Streaming: Server-Sent Events (SSE) for real-time updates
Async Notifications: Webhook-based push notifications for long-running tasks
Security: TLS, OAuth 2.0, API keys, OpenID Connect, Bearer tokens
Discovery Mechanism: Agent Cards

GET https://agent.example.com/.well-known/agent.json

class BFTCoordinatorA2A:
    """Coordinator agent that initiates BFT voting rounds over A2A."""
    
    def __init__(self, agent_id: str, endpoint: str):
        self.agent_id = agent_id
        self.endpoint = endpoint
        self.peers = []  # Registered council agents
        self.vote_rounds = {}
    
    async def discover_council_agents(self):
        """Discover agents via Agent Cards at well-known URIs."""
        council_hosts = [
            "https://agent-1.csoai.network",
            "https://agent-2.csoai.network",
            # ... 47 agents
        ]
        for host in council_hosts:
            card = await self.fetch_agent_card(f"{host}/.well-known/agent.json")
            if "bft-vote" in [s["id"] for s in card.get("skills", [])]:
                self.peers.append({
                    "url": card["url"],
                    "name": card["name"],
                    "sigil_key": card.get("sigil", {}).get("ed25519_pubkey")
                })
    
    async def fetch_agent_card(self, url: str) -> dict:
        """Fetch Agent Card from well-known URI."""
        import aiohttp
        async with aiohttp.ClientSession() as session:
            async with session.get(url) as resp:
                return await resp.json()
    
    async def initiate_vote(self, proposal: dict) -> dict:
        """Initiate a BFT voting round via A2A tasks/send."""
        round_id = f"vote-{datetime.utcnow().isoformat()}"
        
        # Create task messages for each council agent
        tasks = []
        for peer in self.peers:
            task = {
                "jsonrpc": "2.0",
                "method": "tasks/send",
                "params": {
                    "message": {
                        "role": "user",
                        "parts": [{
                            "type": "data",
                            "data": {
                                "type": "bft_vote_request",
                                "round_id": round_id,
                                "proposal": proposal,
                                "quorum": len(self.peers) * 2 // 3 + 1,
                                "deadline": (datetime.utcnow().timestamp() + 300)
                            }
                        }]
                    }
                },
                "id": f"task-{peer['name']}"
            }
            tasks.append(self.send_a2a_task(peer["url"], task))
        
        # Collect votes from all peers
        responses = await asyncio.gather(*tasks, return_exceptions=True)
        votes = self._tally_votes(responses)
        
        return {
            "round_id": round_id,
            "proposal_hash": self._hash_proposal(proposal),
            "votes": votes,
            "result": "approved" if votes["approve"] >= len(self.peers) * 2 // 3 + 1 else "rejected",
            "quorum_reached": votes["total"] >= len(self.peers) * 2 // 3 + 1
        }
    
    async def send_a2a_task(self, endpoint: str, task: dict) -> dict:
        """Send A2A task via JSON-RPC over HTTP."""
        import aiohttp
        async with aiohttp.ClientSession() as session:
            async with session.post(
                endpoint,
                json=task,
                headers={"Authorization": f"Bearer {self._get_token()}",
                        "Content-Type": "application/json"}
            ) as resp:
                return await resp.json()
    
    def _tally_votes(self, responses):
        """Tally Byzantine votes with signature verification."""
        votes = {"approve": 0, "reject": 0, "abstain": 0, "total": 0}
        for r in responses:
            if isinstance(r, Exception):
                continue  # Treat failed responses as abstentions (Byzantine tolerance)
            vote_data = r.get("result", {}).get("status", {}).get("message", {}).get("parts", [{}])[0].get("data", {})
            vote = vote_data.get("vote", "abstain")
            # Verify Ed25519 signature on vote
            if self._verify_vote_signature(vote_data):
                votes[vote] = votes.get(vote, 0) + 1
                votes["total"] += 1
        return votes
    
    def _hash_proposal(self, proposal: dict) -> str:
        import hashlib
        return hashlib.sha256(json.dumps(proposal, sort_keys=True).encode()).hexdigest()

class BFTParticipantA2A:
    """Agent that receives and responds to BFT vote requests over A2A."""
    
    def __init__(self, agent_id: str, private_key: str):
        self.agent_id = agent_id
        self.private_key = private_key  # Ed25519 private key
    
    async def handle_vote_request(self, request: dict) -> dict:
        """Handle incoming A2A vote request and return signed response."""
        message_parts = request["params"]["message"]["parts"]
        vote_request = next(p["data"] for p in message_parts if p["type"] == "data")
        
        # Validate proposal
        if not self._validate_proposal(vote_request["proposal"]):
            return self._build_vote_response(vote_request["round_id"], "reject")
        
        # Check voting power based on Sigil reputation
        voting_power = self._get_voting_power()
        
        # Cast vote (approve if proposal meets criteria)
        vote = "approve" if self._evaluate_proposal(vote_request["proposal"]) else "reject"
        
        return self._build_vote_response(vote_request["round_id"], vote, voting_power)
    
    def _build_vote_response(self, round_id: str, vote: str, power: int = 1) -> dict:
        """Build A2A-compliant vote response with Ed25519 signature."""
        import base64
        from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PrivateKey
        
        vote_payload = {
            "round_id": round_id,
            "agent_id": self.agent_id,
            "vote": vote,
            "voting_power": power,
            "timestamp": datetime.utcnow().isoformat()
        }
        
        # Sign with Ed25519 Sigil key
        signature = self._sign_vote(vote_payload)
        vote_payload["signature"] = base64.b64encode(signature).decode()
        
        return {
            "jsonrpc": "2.0",
            "result": {
                "id": f"vote-{round_id}-{self.agent_id}",
                "status": {
                    "state": "completed",
                    "message": {
                        "role": "assistant",
                        "parts": [{"type": "data", "data": vote_payload}]
                    }
                }
            },
            "id": round_id
        }
    
    def _sign_vote(self, payload: dict) -> bytes:
        from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PrivateKey
        from cryptography.hazmat.primitives import serialization
        private_key = Ed25519PrivateKey.from_private_bytes(bytes.fromhex(self.private_key))
        return private_key.sign(json.dumps(payload, sort_keys=True).encode())

3. Anthropic MCP (Model Context Protocol)
3.1 Overview
The Model Context Protocol (MCP) is an open-source standard for connecting AI applications to external systems. Created by Anthropic in November 2024, it has become the dominant protocol for model-to-tool integration, with explosive ecosystem growth.
3.2 How MCP Works
MCP uses a client-server architecture where: - Host: The LLM application (Claude Desktop, Cursor, VS Code) - Client: Connector within the host that speaks MCP - Server: Service that exposes tools, resources, and prompts
Host (LLM App)
  --> Client (MCP connector)
      --> Server (tool provider)
          --> External API/Database/File
Core Primitives
Transport Options
stdio (85% of servers) – Local process communication
Streamable HTTP (recommended for production) – Stateless/stateful HTTP
SSE (legacy, deprecated) – Server-Sent Events
WebSocket – Bidirectional (emerging)

@mcp.tool()
async def cast_vote(
    proposal_id: str,
    vote: Literal["approve", "reject", "abstain"],
    voter_sigil: str,
    ctx: Context
) -> VoteResponse:
    """
    Cast a signed vote on an active BFT proposal.
    
    Args:
        proposal_id: The ID of the proposal to vote on
        vote: Your vote - approve, reject, or abstain
        voter_sigil: Your Ed25519 public key identifier
    
    Returns:
        A signed VoteResponse with cryptographic proof
    """
    await ctx.info(f"Processing vote from {voter_sigil} on proposal {proposal_id}")
    
    # Verify voter is registered council member
    if not await _is_council_member(voter_sigil):
        raise ValueError(f"Sigil {voter_sigil} is not a registered council member")
    
    # Check proposal is active
    proposal = await _get_proposal(proposal_id)
    if proposal["status"] != "active":
        raise ValueError(f"Proposal {proposal_id} is not active")
    
    # Get voting power from reputation
    voting_power = await _get_voting_power(voter_sigil)
    
    # Sign the vote
    vote_data = VoteResponse(
        proposal_id=proposal_id,
        voter_sigil=voter_sigil,
        vote=vote,
        voting_power=voting_power,
        signature="",  # Will be filled
        timestamp=_now_iso()
    )
    signature = await _sign_with_sigil(vote_data.model_dump())
    vote_data.signature = signature
    
    # Store vote
    await _store_vote(vote_data)
    
    await ctx.info(f"Vote recorded: {vote} with power {voting_power}")
    return vote_data

4. ACP: Agent Communication Protocol
Important: There are multiple protocols using the acronym “ACP” in the agent interoperability space. This section covers all of them.
4.1 IBM’s Agent Communication Protocol (ACP)
IBM’s ACP is a REST-native protocol for agent-to-agent communication, part of the open-source BeeAI ecosystem.
Key Design Decisions
Why IBM Created ACP Instead of Using MCP
IBM initially explored adapting MCP for multi-agent communication but encountered architectural limitations:
Streaming: MCP supports basic message streaming but not fine-grained “delta” updates (tokens, trajectory updates)
Memory Sharing: MCP doesn’t support running multiple agents across servers with shared memory
Message Structure: MCP accepts any JSON schema but doesn’t define message body structure, making interoperability difficult
Protocol Complexity: MCP’s JSON-RPC requires specific SDKs; ACP’s REST design is more integration-friendly

"""
CSOAI BFT Council Governance Protocol
Custom protocol built on top of A2A + MCP
"""

class CSOAIGovernanceProtocol:
    """
    Implements Byzantine Fault Tolerant consensus for CSOAI.
    
    Built on:
    - A2A for agent-to-agent vote message transport
    - MCP for tool-based vote casting and tallying
    - Ed25519 Sigil for identity and signatures
    - 2/3 + 1 quorum for Byzantine fault tolerance
    """
    
    QUORUM_THRESHOLD = 2/3  # 2/3 + 1 for BFT
    TOTAL_AGENTS = 47
    
    def __init__(self):
        self.quorum_size = int(self.TOTAL_AGENTS * self.QUORUM_THRESHOLD) + 1
        # 32 agents required for consensus (2/3 of 47 + 1)
    
    # ===== Voting Round Lifecycle =====
    
    async def start_voting_round(self, proposal: Proposal) -> VotingRound:
        """
        1. Coordinator creates proposal hash
        2. Sends A2A tasks/send to all 47 agents
        3. Collects signed votes via A2A responses
        4. Verifies Ed25519 signatures
        5. Tallies votes
        6. Determines outcome (approve/reject/pending)
        """
        round = VotingRound(proposal=proposal)
        
        # Broadcast via A2A
        responses = await self.a2a_broadcast(
            method="tasks/send",
            message=self._build_vote_request(round)
        )
        
        # Collect and verify
        for response in responses:
            vote = self._extract_vote(response)
            if self.sigil.verify(vote.signature, vote.voter_sigil):
                round.add_vote(vote)
        
        # Check quorum
        if round.total_votes >= self.quorum_size:
            round.outcome = self._determine_outcome(round.votes)
        
        return round
    
    async def submit_vote(self, vote_request: dict) -> SignedVote:
        """
        Agent receives A2A vote request, evaluates, signs, returns.
        Uses MCP tools for validation (get_council_status, etc.)
        """
        # Validate proposal via MCP
        status = await self.mcp_client.call_tool("get_council_status", {})
        
        # AI evaluation
        decision = await self.llm.evaluate(vote_request["proposal"])
        
        # Sign with Ed25519 Sigil
        vote = SignedVote(
            round_id=vote_request["round_id"],
            vote=decision.vote,
            voter_sigil=self.sigil.pubkey,
            signature=self.sigil.sign(decision.to_bytes())
        )
        
        return vote
    
    def verify_consensus(self, round: VotingRound) -> bool:
        """
        Verify that consensus was reached with valid BFT guarantees.
        - At least 32 votes (2/3 + 1)
        - All signatures valid
        - No double-voting
        """
        if round.total_votes < self.quorum_size:
            return False
        
        sigils_seen = set()
        for vote in round.votes:
            if not self.sigil.verify(vote.signature, vote.voter_sigil):
                return False
            if vote.voter_sigil in sigils_seen:
                return False  # Double vote
            sigils_seen.add(vote.voter_sigil)
        
        return True
