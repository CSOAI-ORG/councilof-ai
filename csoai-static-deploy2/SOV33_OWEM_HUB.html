Anatomy of one SIGIL

struct SIGIL {
  version        : u8   // = 1
  prev_hash      : [u8; 64]   // SHA-512 of previous SIGIL · forms chain
  payload_hash   : [u8; 64]   // SHA-512 of (prompt | context | output | metadata)
  agent_did      : str         // "did:csoai:nicholas-001"
  bft_tally      : [u8; 6]    // approve | amend | reject bytes
  care_score     : f32         // MUST be >= 0.95 to emit
  ts_unix_ms     : u64
  sigil_type     : enum{boot, charter, seed, rho, bft, cycle}
  signature      : [u8; 64]    // Ed25519 over all above
}
