5. ROUTING SCRIPT — 4 STEPS RUNNABLE IN <60S

# On a fresh VM (Linux 4.x, bash 5+), run:
$ bash defoneos-mod-procurement-route.sh

#!/bin/bash
# Routing script: maps a buyer's framework-eligibility scope
echo '[1/4] — buyer scope probe'
echo '  Buyer type (T1 / T2 / T3)? '
read tier
echo '  Buyer scope (single / multi-region / air-gap)? '
read scope

echo '[2/4] — framework eligibility filter'
case $tier in
  T1) fws="F3 F4 F9 F10";;
  T2) fws="F1 F2 F3 F4 F9 F10 F11 F14";;
  T3) fws="F1 F2 F3 F4 F5 F6 F7 F8 F9 F10 F11 F12 F13 F14";;
esac
echo "  Eligible frameworks: $fws"

echo '[3/4] — evidence pack retrieval'
for f in $fws; do
  case $f in
    F1) echo '  F1 JSP 936 → /defoneos-buyer-due-diligence-pack.html';;
    F2) echo '  F2 JSP 440 → /defoneos-artefact-tier-cross-reference.html';;
    F3) echo '  F3 G-Cloud 14 → /defoneos-per-tier-pricing-detail.html';;
    *) echo "  $f → see /defoneos-mod-framework-agreement-checklist.html §3";;
  esac
done

echo '[4/4] — readiness probe'
curl -sS -o /dev/null -w 'homepage %{http_code}
' https://www.csoai.org/
curl -sS -o /dev/null -w 'DD pack %{http_code}
' https://www.csoai.org/defoneos-buyer-due-diligence-pack.html
curl -sS -o /dev/null -w 'DD Q-by-Q %{http_code}
' https://www.csoai.org/defoneos-buyer-due-diligence-question-by-question.html
curl -sS -o /dev/null -w 'framework %{http_code}
' https://www.csoai.org/defoneos-mod-framework-agreement-checklist.html
echo '4/4 PASSED'
